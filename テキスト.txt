<script type="module">
    let items = [];
    let currentFilter = 'ALL';

    window.addEventListener('DOMContentLoaded', () => {
        loadLocalStorageFallback();
    });

    function loadLocalStorageFallback() {
        document.getElementById('cloud-status-badge').innerHTML = '<span class="w-1.5 h-1.5 rounded-full bg-amber-500"></span> ローカル保存中';
        const saved = localStorage.getItem('expiry_items_v10');
        if (saved) {
            try { items = JSON.parse(saved); } catch (e) { items = []; }
        } else {
            items = [];
        }
        renderAll();
    }

    function getOffsetDate(days) {
        const d = new Date();
        d.setDate(d.getDate() + days);
        return d.toISOString().split('T')[0];
    }

    async function saveItemToCloud(itemData) {
        const id = itemData.id || crypto.randomUUID();
        const index = items.findIndex(i => i.id === id);
        if (index >= 0) items[index] = { ...itemData, id };
        else items.push({ ...itemData, id });
        localStorage.setItem('expiry_items_v10', JSON.stringify(items));
        renderAll();
    }

    window.deleteItem = async function(id) {
        items = items.filter(i => i.id !== id);
        localStorage.setItem('expiry_items_v10', JSON.stringify(items));
        renderAll();
        showMsg("削除完了", "アイテムを削除しました。", "success");
    }

    window.openModal = function(item = null) {
        const modal = document.getElementById('item-modal');
        const title = document.getElementById('modal-title');
        const form = document.getElementById('item-form');
        form.reset();
        document.getElementById('item-id').value = '';
        document.getElementById('form-no-expiry').checked = false;
        document.getElementById('form-expiry').disabled = false;
        document.getElementById('form-expiry').classList.remove('bg-slate-100', 'text-slate-400', 'cursor-not-allowed');

        const todayStr = new Date().toISOString().split('T')[0];

        if (item && item.id) {
            title.innerText = "アイテムの編集";
            document.getElementById('item-id').value = item.id;
            document.getElementById('form-name').value = item.name || '';
            document.getElementById('form-category').value = item.category || '食品';
            document.getElementById('form-opened').value = item.opened || todayStr;
            
            if (item.expiry === 'なし' || !item.expiry) {
                document.getElementById('form-no-expiry').checked = true;
                document.getElementById('form-expiry').value = '';
                document.getElementById('form-expiry').disabled = true;
                document.getElementById('form-expiry').classList.add('bg-slate-100', 'text-slate-400', 'cursor-not-allowed');
            } else {
                document.getElementById('form-expiry').value = item.expiry;
            }

            document.getElementById('form-memo').value = item.memo || '';
        } else {
            title.innerText = "新規アイテム登録";
            document.getElementById('form-opened').value = todayStr;
            document.getElementById('form-expiry').value = getOffsetDate(7);
        }
        modal.classList.remove('hidden');
    }

    window.toggleNoExpiry = function(checkbox) {
        const expiryInput = document.getElementById('form-expiry');
        if (checkbox.checked) {
            expiryInput.value = '';
            expiryInput.disabled = true;
            expiryInput.required = false;
            expiryInput.classList.add('bg-slate-100', 'text-slate-400', 'cursor-not-allowed');
        } else {
            expiryInput.disabled = false;
            expiryInput.required = false;
            expiryInput.value = getOffsetDate(7);
            expiryInput.classList.remove('bg-slate-100', 'text-slate-400', 'cursor-not-allowed');
        }
    }

    window.closeModal = function() {
        document.getElementById('item-modal').classList.add('hidden');
    }

    window.saveItem = async function(event) {
        event.preventDefault();
        const id = document.getElementById('item-id').value;
        const name = document.getElementById('form-name').value;
        const category = document.getElementById('form-category').value;
        const opened = document.getElementById('form-opened').value;
        const noExpiry = document.getElementById('form-no-expiry').checked;
        const expiry = noExpiry ? 'なし' : document.getElementById('form-expiry').value;
        const memo = document.getElementById('form-memo').value;

        const itemData = { name, category, opened, expiry, memo };
        if (id) itemData.id = id;

        await saveItemToCloud(itemData);
        closeModal();
        showMsg("保存完了", "アイテムを保存しました。", "success");
    }

    window.filterByStatus = function(status) {
        currentFilter = status;
        const banner = document.getElementById('active-filter-banner');
        const text = document.getElementById('active-filter-text');
        if (status === 'ALL') {
            banner.classList.add('hidden');
        } else {
            banner.classList.remove('hidden');
            const mapNames = { SAFE: '安全なアイテム', WARNING: 'まもなく期限のアイテム', EXPIRED: '期限切れアイテム' };
            text.innerText = `フィルター: ${mapNames[status] || status}`;
        }
        renderItems();
    }

    function calculateItemState(item) {
        const today = new Date();
        today.setHours(0,0,0,0);

        if (item.expiry && item.expiry !== 'なし') {
            const expiryDate = new Date(item.expiry);
            const diffDays = Math.ceil((expiryDate - today) / (1000 * 60 * 60 * 24));
            if (diffDays < 0) return { type: 'EXPIRED', label: `期限切れ` };
            if (diffDays <= 3) return { type: 'WARNING', label: `あと ${diffDays} 日` };
            return { type: 'SAFE', label: `あと ${diffDays} 日` };
        }

        return { type: 'SAFE', label: '賞味期限なし' };
    }

    function renderAll() {
        let total = items.length;
        let safeCount = 0;
        let warningCount = 0;
        let expiredCount = 0;

        items.forEach(item => {
            const state = calculateItemState(item);
            if (state.type === 'SAFE') safeCount++;
            else if (state.type === 'WARNING') warningCount++;
            else if (state.type === 'EXPIRED') expiredCount++;
        });

        document.getElementById('stat-total').innerText = total;
        document.getElementById('stat-safe').innerText = safeCount;
        document.getElementById('stat-warning').innerText = warningCount;
        document.getElementById('stat-expired').innerText = expiredCount;

        renderItems();
    }

    window.renderItems = function() {
        const tbody = document.getElementById('items-table-body');
        const searchVal = document.getElementById('search-input').value.toLowerCase();
        const sortVal = document.getElementById('sort-select').value;

        let filtered = items.filter(item => {
            const matchSearch = item.name.toLowerCase().includes(searchVal) || (item.memo && item.memo.toLowerCase().includes(searchVal)) || item.category.toLowerCase().includes(searchVal);
            if (!matchSearch) return false;

            if (currentFilter === 'ALL') return true;
            const state = calculateItemState(item);
            return state.type === currentFilter;
        });

        filtered.sort((a, b) => {
            if (sortVal === 'expiry-asc') return (a.expiry || '9999-12-31').localeCompare(b.expiry || '9999-12-31');
            if (sortVal === 'expiry-desc') return (b.expiry || '').localeCompare(a.expiry || '');
            if (sortVal === 'name-asc') return a.name.localeCompare(b.name);
            return 0;
        });

        if (filtered.length === 0) {
            tbody.innerHTML = `
                <tr>
                    <td colspan="5" class="py-12 text-center text-slate-400">
                        <div class="flex flex-col items-center justify-center space-y-2">
                            <svg class="w-8 h-8 text-slate-300" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M20 13V6a2 2 0 00-2-2H6a2 2 0 00-2 2v7m16 0v5a2 2 0 01-2 2H6a2 2 0 01-2-2v-5m16 0h-2.586a1 1 0 00-.707.293l-2.414 2.414a1 1 0 01-.707.293h-3.172a1 1 0 01-.707-.293l-2.414-2.414A1 1 0 006.586 13H4"></path>
                            </svg>
                            <span class="text-sm font-medium text-slate-600">登録されているアイテムはありません</span>
                        </div>
                    </td>
                </tr>
            `;
            return;
        }

        let html = '';
        filtered.forEach(item => {
            const state = calculateItemState(item);

            let badgeClass = 'bg-emerald-50 text-emerald-700 border-emerald-200';
            if (state.type === 'WARNING') {
                badgeClass = 'bg-amber-50 text-amber-700 border-amber-200 animate-pulse';
            } else if (state.type === 'EXPIRED') {
                badgeClass = 'bg-rose-50 text-rose-700 border-rose-200';
            }

            let elapsedDaysNum = 0;
            if (item.opened) {
                const today = new Date();
                today.setHours(0,0,0,0);
                const openedDate = new Date(item.opened);
                const diffTime = today - openedDate;
                const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));
                elapsedDaysNum = diffDays >= 0 ? diffDays : 0;
            }

            let elapsedBadgeClass = 'bg-emerald-50 text-emerald-700 border-emerald-200';
            if (state.type === 'WARNING') {
                elapsedBadgeClass = 'bg-amber-50 text-amber-700 border-amber-200';
            } else if (state.type === 'EXPIRED') {
                elapsedBadgeClass = 'bg-rose-50 text-rose-700 border-rose-200';
            }
            const elapsedBadgeHtml = `<span class="px-3 py-1 text-base font-bold rounded-lg border ${elapsedBadgeClass}">${elapsedDaysNum}</span>`;

            html += `
                <tr class="hover:bg-slate-50/80 transition-colors">
                    <td class="py-3.5 px-4 font-bold text-slate-900 text-lg whitespace-nowrap">${item.name}</td>
                    <td class="py-3.5 px-4 text-sm text-slate-600 whitespace-nowrap">${item.opened || '-'}</td>
                    <td class="py-3.5 px-4 whitespace-nowrap">${elapsedBadgeHtml}</td>
                    <td class="py-3.5 px-4 whitespace-nowrap">
                        <span class="px-3 py-1 text-sm font-bold rounded-lg border ${badgeClass}">
                            ${state.label}
                        </span>
                    </td>
                    <td class="py-3.5 px-4 text-center whitespace-nowrap">
                        <div class="inline-flex items-center justify-center gap-1.5">
                            <button onclick='openModal(${JSON.stringify(item)})' class="px-3 py-1.5 bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-medium rounded-lg transition-colors">編集</button>
                            <button onclick="deleteItem('${item.id}')" class="px-3 py-1.5 bg-rose-50 hover:bg-rose-100 text-rose-600 text-xs font-medium rounded-lg transition-colors">削除</button>
                        </div>
                    </td>
                </tr>
            `;
        });
        tbody.innerHTML = html;
    }
